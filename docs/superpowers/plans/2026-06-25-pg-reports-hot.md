**Recipe** (do not modify): `services/pgDb.js` → `makePgDb(url)` → `{connect, exec, prepare(sql).get/all/run (async), runTransaction(async fn), close}`; `?`→`$N` + named `@name` (single object → named binding); NUMERIC/BIGINT→JS number. Mirror merged 4C forks + `tools/pg/copy-reports-reports.js` + `tests/pg-reports-reports.test.js`.

**Sync-dispatch invariant:** every forked fn returns synchronously on sqlite (value/count, not Promise); only the pg branch is async. First statement: `if (REPORTS_BACKEND === 'postgres') return _fnPg(...);`. Writers are `const x = db.transaction(...)` → convert to dispatching `function x()` + renamed `const _xSqlite = db.transaction(...)`.

**Dialect:** `datetime('now')`→`now()`; `excluded.col`→`EXCLUDED.col`; `db.transaction()()`→`await runTransaction`. The `INSERT … ON CONFLICT` writers get an auto-appended `RETURNING *` from the adapter (harmless — writers count via `.changes`/rowCount).

function upsertRequestBatch(rows) {
  if (REPORTS_BACKEND === 'postgres') return _upsertRequestBatchPg(rows);
  return _upsertRequestBatchSqlite(rows);
}
const _upsertRequestBatchSqlite = db.transaction((rows) => { let n=0; for (const r of rows){ _upsertRequest.run(r); n++; } return n; });
// SQL = the same INSERT…ON CONFLICT body as _upsertRequest but datetime('now')→now(), excluded.→EXCLUDED.
const _REQ_UPSERT_PG = `INSERT INTO request_records (... ) VALUES (@... , now()) ON CONFLICT(request_id) DO UPDATE SET ... = EXCLUDED. ..., fetched_at = now()`;
async function _upsertRequestBatchPg(rows) {
  const H = pgHandle(); let n = 0;
  await H.runTransaction(async () => { for (const r of (rows||[])) { await H.prepare(_REQ_UPSERT_PG).run(r); n++; } });
  return n;
}
